```python
from __future__ import annotations

from dataclasses import dataclass
from decimal import Decimal, InvalidOperation

CURRENCIES = frozenset({"USD", "USDC", "EUR"})
TYPES = frozenset({"sale", "refund", "chargeback"})
REQUIRED = ("id", "merchant_id", "type", "amount", "currency", "timestamp")


class InvalidRecord(Exception):
    def __init__(self, record_id, reason):
        super().__init__(f"{record_id}: {reason}")
        self.record_id = record_id
        self.reason = reason


@dataclass(frozen=True)
class Txn:
    id: str
    merchant_id: str
    type: str
    amount: Decimal
    currency: str
    timestamp: str


def parse(raw: dict) -> Txn:
    rid = raw.get("id", "<no id>")
    missing = [k for k in REQUIRED if raw.get(k) in (None, "")]
    if missing:
        raise InvalidRecord(rid, f"missing {', '.join(missing)}")
    if raw["type"] not in TYPES:
        raise InvalidRecord(rid, f"unknown type {raw['type']!r}")
    if raw["currency"] not in CURRENCIES:
        raise InvalidRecord(rid, f"unsupported currency {raw['currency']!r}")
    try:
        amount = Decimal(str(raw["amount"]))
    except InvalidOperation:
        raise InvalidRecord(rid, f"bad amount {raw['amount']!r}")
    if amount < 0:
        raise InvalidRecord(rid, f"amount must be positive, got {amount}")
    return Txn(rid, raw["merchant_id"], raw["type"], amount, raw["currency"], raw["timestamp"])
```

```python
from __future__ import annotations

from decimal import ROUND_DOWN, Decimal

from settlement.records import Txn

CENT = Decimal("0.01")

SCHEDULE = {
    "sale": (Decimal("0.029"), Decimal("0.30")),
    "refund": (Decimal("0"), Decimal("0")),
    "chargeback": (Decimal("0"), Decimal("15.00")),
}


def fee_for(txn: Txn) -> Decimal:
    pct, fixed = SCHEDULE[txn.type]
    return (txn.amount * pct + fixed).quantize(CENT, rounding=ROUND_DOWN)
```

--- settlement/aggregate.py ---
```python
from __future__ import annotations

from collections import defaultdict
from dataclasses import dataclass
from decimal import Decimal
from typing import Iterable

from settlement.fees import fee_for
from settlement.records import Txn

ZERO = Decimal("0")


@dataclass
class Totals:
    sales: Decimal = ZERO
    refunds: Decimal = ZERO
    chargebacks: Decimal = ZERO
    fees: Decimal = ZERO

    @property
    def gross(self) -> Decimal:
        return self.sales - self.refunds - self.chargebacks

    @property
    def net(self) -> Decimal:
        return self.gross - self.fees


_BUCKET = {"sale": "sales", "refund": "refunds", "chargeback": "chargebacks"}


def totals_by_merchant(txns: Iterable[Txn]) -> dict[tuple[str, str], Totals]:
    out: dict[tuple[str, str], Totals] = defaultdict(Totals)
    for t in txns:
        key = (t.merchant_id, "ALL")
        bucket = _BUCKET[t.type]
        setattr(out[key], bucket, getattr(out[key], bucket) + t.amount)
        out[key].fees += fee_for(t)
    return dict(out)
```

--- settlement/report.py ---
```python
from __future__ import annotations

from dataclasses import dataclass, field
from decimal import ROUND_HALF_UP, Decimal

from settlement.aggregate import totals_by_merchant
from settlement.records import InvalidRecord, parse

CENT = Decimal("0.01")


@dataclass
class Line:
    merchant_id: str
    currency: str
    sales: Decimal
    refunds: Decimal
    chargebacks: Decimal
    fees: Decimal
    payout: Decimal
    held: bool


@dataclass
class Report:
    lines: list[Line] = field(default_factory=list)
    errors: list[tuple[str, str]] = field(default_factory=list)

    def line(self, merchant_id: str, currency: str) -> Line:
        for ln in self.lines:
            if (ln.merchant_id, ln.currency) == (merchant_id, currency):
                return ln
        raise KeyError((merchant_id, currency))


def _q(d: Decimal) -> Decimal:
    return d.quantize(CENT, rounding=ROUND_HALF_UP)


def settle(raw_records: list[dict]) -> Report:
    report = Report()
    txns = []
    for raw in raw_records:
        try:
            txns.append(parse(raw))
        except InvalidRecord:
            continue

    for (merchant, currency), t in sorted(totals_by_merchant(txns).items()):
        held = t.gross < 0
        report.lines.append(
            Line(
                merchant_id=merchant,
                currency=currency,
                sales=_q(t.sales),
                refunds=_q(t.refunds),
                chargebacks=_q(t.chargebacks),
                fees=_q(t.fees),
                payout=Decimal("0.00") if held else _q(t.net),
                held=held,
            )
        )
    return report
```

```python
from tests.helpers import rec


def test_single_merchant_payout():
```
